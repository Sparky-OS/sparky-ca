cmake_minimum_required(VERSION 3.25)
project(sparky-ca VERSION 2.0.0 LANGUAGES CXX)
if(CMAKE_INSTALL_PREFIX_INITIALIZED_TO_DEFAULT)
    set(CMAKE_INSTALL_PREFIX /usr CACHE PATH "Install prefix" FORCE)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(GNUInstallDirs)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Network)
find_package(KF6I18n REQUIRED)
find_package(OpenSSL 3 REQUIRED)
find_package(LibArchive REQUIRED)
add_library(sparky-ca-core STATIC
    src/core/certificate.cpp src/core/catalogue.cpp src/core/download.cpp src/core/stores.cpp
    src/core/personal.cpp src/core/systemchange.cpp)
target_include_directories(sparky-ca-core PUBLIC src)
target_link_libraries(sparky-ca-core PUBLIC Qt6::Core Qt6::Network KF6::I18n OpenSSL::Crypto LibArchive::LibArchive)
target_compile_options(sparky-ca-core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(sparky-ca src/cli/main.cpp)
target_link_libraries(sparky-ca PRIVATE sparky-ca-core)
target_compile_definitions(sparky-ca PRIVATE SPARKY_CA_DATA="${CMAKE_INSTALL_FULL_DATADIR}/sparky-ca")
install(TARGETS sparky-ca RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(DIRECTORY data/ DESTINATION ${CMAKE_INSTALL_DATADIR}/sparky-ca)
include(CTest)
option(BUILD_ED521_PROVIDER "Build the public-verification Ed521 prototype" OFF)
if(BUILD_ED521_PROVIDER)
    add_subdirectory(ed521-provider)
endif()
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_test(NAME core-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:sparky-ca>)
    add_test(NAME e521-integrity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/e521.py $<TARGET_FILE:sparky-ca>)
    set_tests_properties(core-integration PROPERTIES TIMEOUT 120)
    set_tests_properties(e521-integrity PROPERTIES TIMEOUT 120)
    if(BUILD_ED521_PROVIDER)
        add_test(NAME core-ed521-provider COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/provider.py $<TARGET_FILE:sparky-ca> $<TARGET_FILE:ed521-provider>)
        set_tests_properties(core-ed521-provider PROPERTIES TIMEOUT 120)
    endif()
endif()

if(BUILD_TESTING)
    add_executable(system-request-check tests/system-request.cpp)
    target_link_libraries(system-request-check PRIVATE sparky-ca-core)
    find_file(P11KIT_PROXY_LIBRARY NAMES p11-kit-proxy.so PATHS ${CMAKE_PREFIX_PATH} /usr PATH_SUFFIXES lib/${CMAKE_LIBRARY_ARCHITECTURE} lib)
    find_file(OPENSC_MODULE_LIBRARY NAMES opensc-pkcs11.so PATHS ${CMAKE_PREFIX_PATH} /usr PATH_SUFFIXES lib/${CMAKE_LIBRARY_ARCHITECTURE} lib/${CMAKE_LIBRARY_ARCHITECTURE}/pkcs11 lib/pkcs11)
    if(P11KIT_PROXY_LIBRARY AND OPENSC_MODULE_LIBRARY)
        add_test(NAME token-module COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/token-module.py $<TARGET_FILE:sparky-ca> ${P11KIT_PROXY_LIBRARY} ${OPENSC_MODULE_LIBRARY})
        set_tests_properties(token-module PROPERTIES TIMEOUT 60)
    endif()
endif()
