add_library(sparky-ca-core STATIC
    src/core/certificate.cpp src/core/catalogue.cpp src/core/download.cpp src/core/stores.cpp
    src/core/personal.cpp src/core/systemchange.cpp src/core/revocation.cpp src/core/automation.cpp src/core/ed521.cpp)

target_link_libraries(sparky-ca-core PUBLIC Qt6::Core Qt6::Network KF6::I18n OpenSSL::Crypto LibArchive::LibArchive)

install(DIRECTORY data/ DESTINATION ${CMAKE_INSTALL_DATADIR}/sparky-ca)
install(FILES share/sparky-ca.1 DESTINATION ${CMAKE_INSTALL_MANDIR}/man1)
find_program(MSGFMT_PROGRAM msgfmt REQUIRED)
file(STRINGS po/LINGUAS SPARKY_CA_LANGUAGES)
set(SPARKY_CA_TRANSLATED)
foreach(language ${SPARKY_CA_LANGUAGES})
    set(catalog ${CMAKE_CURRENT_BINARY_DIR}/locale/${language}/LC_MESSAGES/sparky-ca.mo)
    add_custom_command(OUTPUT ${catalog}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/locale/${language}/LC_MESSAGES
        COMMAND ${MSGFMT_PROGRAM} --check-format -o ${catalog} ${CMAKE_CURRENT_SOURCE_DIR}/po/${language}.po
        DEPENDS po/${language}.po)
    list(APPEND SPARKY_CA_TRANSLATED ${catalog})
    install(FILES ${catalog} DESTINATION ${CMAKE_INSTALL_DATADIR}/locale/${language}/LC_MESSAGES)
endforeach()
# The Ed521 message has its own domain and ships in all of the edition's languages.
file(STRINGS po/ed521/LINGUAS SPARKY_CA_ED521_LANGUAGES)
string(REPLACE " " ";" SPARKY_CA_ED521_LANGUAGES "${SPARKY_CA_ED521_LANGUAGES}")
foreach(language ${SPARKY_CA_ED521_LANGUAGES})
    set(catalog ${CMAKE_CURRENT_BINARY_DIR}/locale/${language}/LC_MESSAGES/sparky-ca-ed521.mo)
    add_custom_command(OUTPUT ${catalog}
        COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_CURRENT_BINARY_DIR}/locale/${language}/LC_MESSAGES
        COMMAND ${MSGFMT_PROGRAM} --check-format -o ${catalog} ${CMAKE_CURRENT_SOURCE_DIR}/po/ed521/${language}.po
        DEPENDS po/ed521/${language}.po)
    list(APPEND SPARKY_CA_TRANSLATED ${catalog})
    install(FILES ${catalog} DESTINATION ${CMAKE_INSTALL_DATADIR}/locale/${language}/LC_MESSAGES)
endforeach()
# The desktop entries and the AppStream metadata take their translations from the same po files.
foreach(template share/sparky-ca.desktop share/kservices/sparky-ca-preview.desktop)
    get_filename_component(leaf ${template} NAME)
    add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/${leaf}
        COMMAND ${MSGFMT_PROGRAM} --desktop --template ${CMAKE_CURRENT_SOURCE_DIR}/${template}.in -d ${CMAKE_CURRENT_SOURCE_DIR}/po -o ${CMAKE_CURRENT_BINARY_DIR}/${leaf}
        DEPENDS ${template}.in po/LINGUAS)
    list(APPEND SPARKY_CA_TRANSLATED ${CMAKE_CURRENT_BINARY_DIR}/${leaf})
endforeach()
add_custom_command(OUTPUT ${CMAKE_CURRENT_BINARY_DIR}/org.sparkylinux.sparkyca.metainfo.xml
    COMMAND ${MSGFMT_PROGRAM} --xml --template ${CMAKE_CURRENT_SOURCE_DIR}/share/org.sparkylinux.sparkyca.metainfo.xml.in -d ${CMAKE_CURRENT_SOURCE_DIR}/po -o ${CMAKE_CURRENT_BINARY_DIR}/org.sparkylinux.sparkyca.metainfo.xml
    DEPENDS share/org.sparkylinux.sparkyca.metainfo.xml.in po/LINGUAS)
list(APPEND SPARKY_CA_TRANSLATED ${CMAKE_CURRENT_BINARY_DIR}/org.sparkylinux.sparkyca.metainfo.xml)
add_custom_target(translations ALL DEPENDS ${SPARKY_CA_TRANSLATED})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/org.sparkylinux.sparkyca.metainfo.xml DESTINATION ${CMAKE_INSTALL_DATADIR}/metainfo)
