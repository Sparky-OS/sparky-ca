cmake_minimum_required(VERSION 3.20)
project(ed521-provider VERSION 0.2.0 LANGUAGES C)
include(GNUInstallDirs)
find_package(OpenSSL 3.0 REQUIRED COMPONENTS Crypto)
find_package(PkgConfig REQUIRED)
pkg_get_variable(OPENSSL_MODULESDIR libcrypto modulesdir)
if(NOT OPENSSL_MODULESDIR)
    message(FATAL_ERROR "libcrypto.pc names no modulesdir")
endif()
add_library(ed521-provider MODULE src/provider.c src/keymgmt.c src/decoder.c src/signature.c src/verify.c)
target_link_libraries(ed521-provider PRIVATE OpenSSL::Crypto)
target_compile_features(ed521-provider PRIVATE c_std_11)
target_compile_options(ed521-provider PRIVATE -Wall -Wextra -Werror)
# OpenSSL loads a provider named ed521 from MODULESDIR/ed521.so when the configuration names no module.
set_target_properties(ed521-provider PROPERTIES PREFIX "" OUTPUT_NAME ed521 C_VISIBILITY_PRESET hidden)
install(TARGETS ed521-provider LIBRARY DESTINATION ${OPENSSL_MODULESDIR})
install(FILES openssl-ed521.cnf DESTINATION ${CMAKE_INSTALL_DATADIR}/openssl-provider-ed521 RENAME openssl.cnf)
install(FILES ICP-Brasilv7.pem DESTINATION ${CMAKE_INSTALL_DATADIR}/openssl-provider-ed521)
install(PROGRAMS system-config.sh DESTINATION ${CMAKE_INSTALL_LIBEXECDIR}/openssl-provider-ed521 RENAME system-config)

add_executable(ed521-check tests/check.c)
target_link_libraries(ed521-check PRIVATE OpenSSL::Crypto)
target_compile_options(ed521-check PRIVATE -Wall -Wextra -Werror)
add_executable(ed521-inventory tests/inventory.c)
target_link_libraries(ed521-inventory PRIVATE OpenSSL::Crypto)
target_compile_options(ed521-inventory PRIVATE -Wall -Wextra -Werror)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
add_test(NAME provider-integration COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py $<TARGET_FILE:ed521-check> $<TARGET_FILE:ed521-provider>)
set_tests_properties(provider-integration PROPERTIES TIMEOUT 180)
add_test(NAME provider-constraints COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/constraints.py $<TARGET_FILE:ed521-check> $<TARGET_FILE:ed521-provider>)
set_tests_properties(provider-constraints PROPERTIES TIMEOUT 60)
add_test(NAME oracle-specification COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/spec.py)
set_tests_properties(oracle-specification PROPERTIES TIMEOUT 120)
add_test(NAME provider-vectors COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/vectors.py $<TARGET_FILE:ed521-check> $<TARGET_FILE:ed521-provider>)
set_tests_properties(provider-vectors PROPERTIES TIMEOUT 300)
add_test(NAME provider-safety COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/safety.py $<TARGET_FILE:ed521-inventory> $<TARGET_FILE:ed521-provider> ${CMAKE_CURRENT_SOURCE_DIR}/openssl-ed521.cnf)
set_tests_properties(provider-safety PROPERTIES TIMEOUT 120)
add_test(NAME provider-unchanged-default COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/unchanged.py ctest $<TARGET_FILE:ed521-provider> ${CMAKE_CURRENT_SOURCE_DIR}/openssl-ed521.cnf)
set_tests_properties(provider-unchanged-default PROPERTIES TIMEOUT 900)
add_test(NAME provider-chain COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/chain.py ${CMAKE_CURRENT_SOURCE_DIR}/openssl-ed521.cnf $<TARGET_FILE:ed521-provider>)
set_tests_properties(provider-chain PROPERTIES TIMEOUT 300)
# The libFuzzer harnesses, built here with tests/fuzz/replay.c instead of libFuzzer, replay the kept corpus.
set(ED521_PROVIDER_SOURCES src/provider.c src/keymgmt.c src/decoder.c src/signature.c src/verify.c)
foreach(harness spki verify verify-v7 x509-verify)
    string(REPLACE "-v7" "" harness_source ${harness})
    add_executable(ed521-fuzz-${harness}-replay tests/fuzz/fuzz-${harness_source}.c tests/fuzz/replay.c ${ED521_PROVIDER_SOURCES})
    target_include_directories(ed521-fuzz-${harness}-replay PRIVATE src tests/fuzz)
    target_link_libraries(ed521-fuzz-${harness}-replay PRIVATE OpenSSL::Crypto)
    target_compile_options(ed521-fuzz-${harness}-replay PRIVATE -Wall -Wextra -Werror)
    if(harness STREQUAL "verify-v7")
        target_compile_definitions(ed521-fuzz-${harness}-replay PRIVATE ED521_FUZZ_FIXED_V7)
    endif()
    string(REPLACE "x509-verify" "x509" corpus ${harness})
    add_test(NAME fuzz-replay-${harness} COMMAND ed521-fuzz-${harness}-replay ${CMAKE_CURRENT_SOURCE_DIR}/tests/fuzz/corpus/${corpus})
    set_tests_properties(fuzz-replay-${harness} PROPERTIES TIMEOUT 300)
endforeach()
